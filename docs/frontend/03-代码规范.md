### 目录规则

- `views/` — 仅放路由页面组件，一个路由一个 `.vue` 文件
- `components/layout/` — 布局类组件
- `components/common/` — 跨页面复用的业务组件
- `api/` — 每类 API 一个文件，导出纯函数
- `stores/` — 每类状态一个 Pinia store
- `types/` — 纯类型定义，无运行时逻辑

## 三、Vue 组件规范

### 3.1 语法风格

一律使用 `<script setup lang="ts">` + Composition API。

```vue
<script setup lang="ts">
import { ref, computed, onMounted } from 'vue'
import type { WebTunnel } from '@/types/tunnel'

const props = defineProps<{
  clientId?: number
}>()

const emit = defineEmits<{
  created: [tunnel: WebTunnel]
}>()

const loading = ref(false)
const data = ref<WebTunnel[]>([])

const activeCount = computed(() => data.value.filter(t => t.isEnabled).length)

onMounted(async () => {
  await fetchList()
})

async function fetchList() { /* ... */ }
function handleCreate() { /* ... */ }
</script>
```

### 3.2 SFC 区块顺序

```vue
<script setup lang="ts">
// 1. imports
// 2. props / emits / slots
// 3. composables
// 4. reactive state (ref / reactive)
// 5. computed
// 6. watchers
// 7. lifecycle hooks
// 8. methods
</script>

<template>
  <!-- 模板 -->
</template>

<style scoped>
/* 样式 */
</style>
```

### 3.3 组件拆分原则

满足以下任一条件即拆分：

| 条件 | 拆分方式 |
|------|----------|
| 一个 `.vue` 文件超过 **200 行** | 提取子组件或 composable |
| 3 个以上独立 UI 区块（表单、列表、筛选栏） | 各为一个子组件 |
| 逻辑含状态管理 + 副作用 | 提取为 `useXxx()` composable |
| 模板出现重复结构 | 提取为可复用组件 |

**视图组件保持薄**：route-level 的 `views/*.vue` 只做数据获取和子组件编排，不包含大量模板。

### 3.4 Props / Emits

- 使用 TypeScript 泛型定义，不用运行时声明
- Props 只读，不可在子组件中直接修改
- 需要修改时 emit 事件，由父组件处理
- `v-model` 仅用于双向绑定表单值

```ts
// ✅ 推荐
const props = defineProps<{
  tunnelId: number
  disabled?: boolean
}>()

const emit = defineEmits<{
  deleted: [id: number]
  updated: [id: number]
}>()
```

### 3.5 模板规范

- 组件标签使用 `kebab-case`：`<a-table>`, `<tunnel-form>`
- `v-if` / `v-for` 不同时出现在同一元素（用 `<template>` 包裹）
- 列表渲染必须带 `:key`，key 使用唯一 ID 而非 index
- 模板逻辑保持声明式，不写复杂表达式

```vue
<!-- ✅ 推荐 -->
<template v-for="item in list" :key="item.id">
  <a-list-item>{{ item.name }}</a-list-item>
</template>

<!-- ❌ 禁止 -->
<a-list-item v-for="(item, i) in list" :key="i">
  {{ computeSomething(item) }}
</a-list-item>
```

## 五、状态管理 (Pinia)

### 5.1 Store 规范

```ts
// stores/auth.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { ApiResponse } from '@/types/api'

export const useAuthStore = defineStore('auth', () => {
  const token = ref<string | null>(localStorage.getItem('token'))
  const username = ref('')
  const preAuthToken = ref<string | null>(null)
  const mfaBound = ref<boolean | null>(null)

  const isAuthenticated = computed(() => token.value !== null)
  const needsMfa = computed(() => preAuthToken.value !== null && token.value === null)

  function setPreAuth(token: string, name: string, bound: boolean) {
    preAuthToken.value = token
    username.value = name
    mfaBound.value = bound
  }

  function setAuth(newToken: string, name: string) {
    token.value = newToken
    username.value = name
    preAuthToken.value = null
    mfaBound.value = null
    localStorage.setItem('token', newToken)
  }

  function logout() {
    token.value = null
    username.value = ''
    preAuthToken.value = null
    mfaBound.value = null
    localStorage.removeItem('token')
  }

  return { token, username, preAuthToken, mfaBound, isAuthenticated, needsMfa, setPreAuth, setAuth, logout }
})

// stores/setup.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getSetupStatus } from '@/api/setup'

export const useSetupStore = defineStore('setup', () => {
  const initialized = ref<boolean | null>(null)
  const loading = ref(false)

  const needsSetup = computed(() => initialized.value === false)

  async function checkStatus() {
    loading.value = true
    try {
      const res = await getSetupStatus()
      initialized.value = res.data.initialized
    } finally {
      loading.value = false
    }
  }

  function markInitialized() {
    initialized.value = true
  }

  return { initialized, loading, needsSetup, checkStatus, markInitialized }
})
```

### 5.2 Store 使用规则

- 全部使用 **Setup Store** 语法（`defineStore('id', () => {})`）
- 从 store 解构时使用 `storeToRefs()` 保持响应性

```ts
// ✅ 推荐
const authStore = useAuthStore()
const { token, isAuthenticated } = storeToRefs(authStore)
authStore.logout()

// ❌ 禁止 — 失去响应性
const { token } = useAuthStore()
```

## 六、API 层规范

### 6.1 Axios 实例

```ts
// api/request.ts
import axios from 'axios'
import { Message } from '@arco-design/web-vue'
import { useAuthStore } from '@/stores/auth'
import router from '@/router'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000,
})

request.interceptors.request.use((config) => {
  const authStore = useAuthStore()
  if (authStore.token) {
    config.headers.Authorization = `Bearer ${authStore.token}`
  }
  return config
})

request.interceptors.response.use(
  (response) => {
    const { success, message: msg } = response.data
    if (!success) {
      Message.error(msg || '请求失败')
      return Promise.reject(new Error(msg))
    }
    return response.data
  },
  (error) => {
    if (error.response?.status === 401) {
      useAuthStore().logout()
      router.push('/login')
    }
    Message.error(error.message || '网络错误')
    return Promise.reject(error)
  },
)

export default request
```

### 7.2 导航守卫（三层：初始化 → 登录 → 权限）

```ts
router.beforeEach(async (to, _from) => {
  const setupStore = useSetupStore()

  if (!setupStore.initialized && !setupStore.loading) {
    await setupStore.checkStatus()
  }

  // 第一层：未初始化 → 必须去 /setup
  if (!setupStore.initialized) {
    if (to.name !== 'Setup') return { name: 'Setup' }
    return
  }

  // 已初始化后 /setup 不再可访问
  if (to.name === 'Setup') {
    return { name: 'Login' }
  }

  // 第二层：未登录 → /login
  const authStore = useAuthStore()
  if (to.meta.requiresAuth !== false && !authStore.isAuthenticated) {
    return { name: 'Login', query: { redirect: to.fullPath } }
  }

  // 第三层：已登录访问 /login → 跳转 Dashboard
  if (to.name === 'Login' && authStore.isAuthenticated) {
    return { name: 'Dashboard' }
  }
})
```

## 九、Composable 规范

```ts
// composables/usePagination.ts
import { ref, computed } from 'vue'

export function usePagination(defaultPageSize = 20) {
  const page = ref(1)
  const pageSize = ref(defaultPageSize)
  const total = ref(0)

  const pagination = computed(() => ({
    current: page.value,
    pageSize: pageSize.value,
    total: total.value,
    showTotal: true,
    showPageSize: true,
  }))

  function onPageChange(current: number) {
    page.value = current
  }

  function onPageSizeChange(size: number) {
    pageSize.value = size
    page.value = 1
  }

  function reset() {
    page.value = 1
  }

  return { page, pageSize, total, pagination, onPageChange, onPageSizeChange, reset }
})
```

```ts
// composables/useMfaLogin.ts  — 登录页 MFA 流程编排
import { ref, computed } from 'vue'
import { useRouter } from 'vue-router'
import { Message } from '@arco-design/web-vue'
import { useAuthStore } from '@/stores/auth'
import { login } from '@/api/auth'
import { setupMfa, bindMfa, verifyMfa } from '@/api/mfa'

export function useMfaLogin() {
  const router = useRouter()
  const authStore = useAuthStore()

  const form = ref({ name: '', password: '' })
  const totpCode = ref('')
  const step = ref<'credentials' | 'mfaSetup' | 'mfaVerify'>('credentials')

  const secret = ref('')
  const qrCodeUrl = ref('')
  const mfaCountdown = ref(300) // preAuthToken 5分钟有效

  async function handleLogin() {
    try {
      const res = await login({ name: form.value.name, password: form.value.password })
      authStore.setPreAuth(res.data.preAuthToken, res.data.username, res.data.mfaBound)
      startCountdown()

      if (res.data.mfaBound) {
        step.value = 'mfaVerify'
      } else {
        const mfaRes = await setupMfa(res.data.preAuthToken)
        secret.value = mfaRes.data.secret
        qrCodeUrl.value = mfaRes.data.qrCodeUrl
        step.value = 'mfaSetup'
      }
    } catch {
      // 凭据错误，拦截器已 toast
    }
  }

  async function handleBindMfa() {
    try {
      const res = await bindMfa(totpCode.value, authStore.preAuthToken!)
      authStore.setAuth(res.data.token, res.data.username)
      Message.success('MFA 绑定成功')
      router.push('/')
    } catch {
      totpCode.value = ''
    }
  }

  async function handleVerifyMfa() {
    try {
      const res = await verifyMfa(totpCode.value, authStore.preAuthToken!)
      authStore.setAuth(res.data.token, res.data.username)
      router.push('/')
    } catch {
      totpCode.value = ''
    }
  }

  function backToLogin() {
    authStore.logout()
    step.value = 'credentials'
    totpCode.value = ''
  }

  function startCountdown() {
    mfaCountdown.value = 300
    const timer = setInterval(() => {
      mfaCountdown.value--
      if (mfaCountdown.value <= 0) {
        clearInterval(timer)
        backToLogin()
        Message.warning('预认证令牌已过期，请重新登录')
      }
    }, 1000)
  }

  return { form, totpCode, step, secret, qrCodeUrl, mfaCountdown,
    handleLogin, handleBindMfa, handleVerifyMfa, backToLogin }
}
```

### Composables 规则

- 一个文件一个 composable
- 返回对象明确，不返回内部用不到的状态
- 有副作用（API 调用、事件监听）时需要在 `onUnmounted` 清理

## 十一、禁止事项

| 禁止 | 说明 |
|------|------|
| `any` 类型 | 用 `unknown` 替代 |
| Options API | 全项目统一 Composition API |
| `this.$xxx` | Composition API 中不存在 this |
| `require()` | Bun/ESM 环境用 `import` |
| 直接修改 props | 数据流单向 |
| `v-if` + `v-for` 同元素 | 用 `<template>` 包裹 |
| `index` 作 key | 使用唯一 ID |
| 样式硬编码色值 | 使用 Arco Design Token |
| 超过 200 行的 `.vue` | 必须拆分 |
| `setTimeout` 等延迟操作不清除 | 在 `onUnmounted` 中 `clearTimeout` |
